```rust
use std::process::{Child, Command, Stdio};
use std::time::Duration;
use crate::obscura::{ObscuraConfig, ObscuraError};

pub struct ObscuraServer {
    child: Option<Child>,
    cdp_url: String,
    external: bool,
}

impl ObscuraServer {
    pub async fn start(cfg: &ObscuraConfig) -> Result<Self, ObscuraError> {
        let (child, external, base) = match &cfg.url {
            Some(url) => (None, true, url.trim_end_matches('/').to_string()),
            None => {
                if !cfg.bin.exists() {
                    return Err(ObscuraError::BinaryNotFound(cfg.bin.display().to_string()));
                }
                let child = Command::new(&cfg.bin)
                    .args(["serve", "--port", &cfg.port.to_string(), "--host", "127.0.0.1",
                           "--workers", "1", "--user-agent", "football-app-v2"])
                    .stdout(Stdio::null()).stderr(Stdio::null())
                    .spawn()
                    .map_err(|e| ObscuraError::ServerStart(e.to_string()))?;
                (Some(child), false, format!("http://127.0.0.1:{}", cfg.port))
            }
        };
        let cdp_url = format!("{base}/devtools/browser");
        // health-check: /json/version poll (max ~10s, 500ms köz)
        let version_url = format!("{base}/json/version");
        let client = reqwest::Client::new();
        let mut ok = false;
        for _ in 0..20 {
            if let Ok(resp) = client.get(&version_url).timeout(Duration::from_millis(500)).send().await {
                if resp.status().is_success() { ok = true; break; }
            }
            tokio::time::sleep(Duration::from_millis(500)).await;
        }
        if !ok {
            if let Some(mut c) = child { let _ = c.kill(); let _ = c.wait(); }
            return Err(ObscuraError::ServerStart(format!("health-check timeout: {version_url}")));
        }
        Ok(Self { child, cdp_url, external })
    }

    pub fn cdp_url(&self) -> &str { &self.cdp_url }

    pub fn stop(&mut self) {
        if let Some(mut c) = self.child.take() {
            let _ = c.kill();
            let _ = c.wait();
        }
    }
}

impl Drop for ObscuraServer {
    fn drop(&mut self) { self.stop(); }
}
```
Megjegyzés: a reqwest már ingest dep. A health-check reqwest timeout-ja a builder-en: `client.get(...).timeout(...)` — a reqwest RequestBuilder timeout metódus. A spawn Stdio::null() hogy ne blokkoljon a pipe; a naplózás a footballd-ben történik (a child output redirect a footballd logba — egyszerűsítés: null, a v1 is naplózott de a spike null-lal ment).

obscura/mod.rs ObscuraFetcher:
```rust
use chromiumoxide::cdp::browser_protocol::network::EventResponseReceived;
use chromiumoxide::Browser;
use futures::StreamExt;
use crate::fetch::{Fetcher, FetchRequest, FetchLimits, RawPayload};
use football_domain::error::FetchError;
use async_trait::async_trait;

pub struct ObscuraFetcher {
    browser: Browser,
    timeout: std::time::Duration,
}

impl ObscuraFetcher {
    pub async fn connect(cdp_url: &str, timeout: std::time::Duration) -> Result<Self, ObscuraError> {
        let (browser, mut handler) = Browser::connect(cdp_url).await
            .map_err(|e| ObscuraError::Cdp(e.to_string()))?;
        tokio::spawn(async move { while let Some(h) = handler.next().await { if h.is_err() { break; } } });
        Ok(Self { browser, timeout })
    }
}

#[async_trait]
impl Fetcher for ObscuraFetcher {
    async fn fetch(&self, req: &FetchRequest, limits: &FetchLimits) -> Result<RawPayload, FetchError> {
        let fut = async {
            let page = self.browser.new_page("about:blank").await
                .map_err(|e| ObscuraError::Cdp(e.to_string()))?;
            let mut responses = page.event_listener::<EventResponseReceived>().await
                .map_err(|e| ObscuraError::Cdp(e.to_string()))?;
            // Network.enable a státusz-capture-hoz (a page CDP session-jén)
            page.execute(chromiumoxide::cdp::browser_protocol::network::EnableParams::default()).await
                .map_err(|e| ObscuraError::Cdp(e.to_string()))?;
            page.goto(&req.url).await.map_err(|e| ObscuraError::Cdp(e.to_string()))?;
            // státusz-capture a response stream-ből
            let mut status: Option<u16> = None;
            let deadline = tokio::time::Instant::now() + std::time::Duration::from_secs(3);
            while tokio::time::Instant::now() < deadline {
                match tokio::time::timeout(std::time::Duration::from_millis(300), responses.next()).await {
                    Ok(Some(ev)) => {
                        if ev.response.url.trim_end_matches('/') == req.url.trim_end_matches('/') {
                            status = Some(ev.response.status as u16);
                        }
                    }
                    _ => break,
                }
            }
            let http_status = status.ok_or(ObscuraError::NoStatus)?;
            // settle + retry content (Obscura context-id időzítés)
            tokio::time::sleep(std::time::Duration::from_millis(1500)).await;
            let html = {
                let mut r = page.content().await;
                for _ in 0..3 {
                    if r.is_ok() { break; }
                    tokio::time::sleep(std::time::Duration::from_secs(1)).await;
                    r = page.content().await;
                }
                r.map_err(|e| ObscuraError::Cdp(e.to_string()))?
            };
            let bytes = html.into_bytes();
            if bytes.len() as u64 > limits.max_bytes {
                return Err(ObscuraError::Cdp(format!("méretplafon: {} byte", bytes.len())));
            }
            let _ = page.close().await;
            Ok(RawPayload {
                sha256: hex::encode(sha2::Sha256::digest(&bytes)),
                bytes,
                content_type: "text/html".to_string(),
                http_status,
                fetched_at: chrono::Utc::now(),
            })
        };
        match tokio::time::timeout(self.timeout.min(limits.timeout), fut).await {
            Ok(res) => res.map_err(|e: ObscuraError| e.into_fetch()),
            Err(_) => Err(FetchError::Timeout),
        }
    }
}
```
Megjegyzés: a futures crate kell az ingest obscura feature-höz (StreamExt) — `futures = { version = "0.3", optional = true }` + a feature-be "dep:futures". fetch.rs Fetchers:
```rust
use football_domain::policy::Transport;
use crate::http::RustHttpFetcher;

pub struct Fetchers {
    pub rust_http: RustHttpFetcher,
    pub fixture: FixtureFetcher,
    #[cfg(feature = "obscura")]
    pub obscura: Option<crate::obscura::ObscuraFetcher>,
}

impl Fetchers {
    pub fn rust_http(&self) -> &dyn Fetcher { &self.rust_http }
    pub fn for_transport(&self, transport: &Transport) -> Result<&dyn Fetcher, FetchError> {
        match transport {
            Transport::RustHttp => Ok(&self.rust_http),
            Transport::Fixture => Ok(&self.fixture),
            #[cfg(feature = "obscura")]
            Transport::Obscura => self.obscura.as_ref()
                .map(|f| f as &dyn Fetcher)
                .ok_or_else(|| FetchError::Transport("obscura fetcher nincs inicializálva".into())),
            #[cfg(not(feature = "obscura"))]
            Transport::Obscura => Err(FetchError::Transport("obscura feature nincs engedélyezve".into())),
        }
    }
}
```

A content fetch előtt: `let fetcher = fetchers.for_transport(&policy.transport)?;` majd `fetcher.fetch(req, &limits)`. A run_loop továbbadja a fetchers-t a process_job-nak.
